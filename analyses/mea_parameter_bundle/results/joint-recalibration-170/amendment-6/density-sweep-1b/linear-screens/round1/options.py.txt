# Minimize non-density C205 subject to all 12 density residuals in [-1,1], plus explicit box constraints.
def hard_density(A,b,D,d,lo,hi):
 n=len(lo); Gbox=np.r_[np.eye(n),-np.eye(n)];hbox=np.r_[hi,-lo];Gd=np.r_[D,-D];hd=np.r_[1-d,1+d]
 x0=np.clip(np.zeros(n),lo,hi);t=max(0.,float(np.max(abs(d+D@x0))-1))+1e-8
 G=np.r_[np.c_[Gbox,np.zeros(2*n)],np.c_[Gd,-np.ones(len(hd))],np.array([[0.]*n+[-1.]])];h=np.r_[hbox,hd,0.]
 phaseA=np.zeros((1,n+1));phaseA[0,-1]=1
 feas,info=qpls(phaseA,np.zeros(1),G,h,np.r_[x0,t]);x=feas[:-1]
 if feas[-1]>1e-6:return x,{'feasible':False,'min_max_density':float(2*(1+feas[-1])),'phase_kkt':info}
 x,info=qpls(A,b,np.r_[Gbox,Gd],np.r_[hbox,hd],x)
 return x,{'feasible':True,'kkt':info}
