def run(producer, edits, argv, file=None):
    # Execute in __main__ so the producer's fork pools can pickle its functions by name.
    text = (ROOT/producer).read_text()
    for old, new in edits:
        assert old in text, old
        text = text.replace(old, new)
    sys.argv, namespace = [producer, *argv], sys.modules['__main__'].__dict__
    namespace['__file__'] = str(file or ROOT/producer)
    exec(compile(text, producer, 'exec'), namespace)


def gate(family, rows):  # rows: (identity, old, new); delta and zero rule of issue #203
    out = []
    for identity, old, new in rows:
        new = float('nan') if new in (None, '') else float(new)
        old = float(old)
        delta = abs(new-old)/max(abs(old), abs(new), 1e-300) if math.isfinite(new) else math.inf
        out.append(dict(family=family, identity=identity, old=old, new=new, delta=delta,
                        passed=math.isfinite(new) and (delta <= 1e-10 or old == 0 and abs(new) <= 1e-300)))
    return out


def leaves(value, path=''):
    if isinstance(value, dict):
        return [x for k, v in value.items() for x in leaves(v, f'{path}/{k}')]
    if isinstance(value, list):
        return [x for i, v in enumerate(value) for x in leaves(v, f'{path}/{i}')]
    return [(path, value)]


def paired(family, old, new):  # numeric leaves are gated; any other differing leaf fails
    old, new = dict(leaves(old)), dict(leaves(new))  # a leaf present on one side only also fails
    numeric = [k for k in old.keys() & new.keys() if type(old[k]) in (int, float)]
    return gate(family, [(k, old[k], new[k]) for k in numeric]) + [dict(family=family, identity=k, old=old.get(k, 'absent'), new=new.get(k, 'absent'), delta=math.inf,
        passed=False) for k in sorted(old.keys() | new.keys()) if k not in numeric and old.get(k, 'absent') != new.get(k, 'absent')]


def table(path, key, fields):
    return {tuple(r[k] for k in key): {f: float(r[f]) if r[f] not in ('', 'None') else None for f in fields} for r in csv.DictReader(open(path))}
