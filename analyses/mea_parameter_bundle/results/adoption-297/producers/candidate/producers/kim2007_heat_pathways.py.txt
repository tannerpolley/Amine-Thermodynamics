def metrics(rows, predicted='predicted', observed='observed'):
    used = [r for r in rows if r.get(predicted) is not None and r.get(observed) is not None]
    errors = [r[predicted] - r[observed] for r in used]
    relative = [abs((r[predicted] - r[observed]) / r[observed]) for r in used if r[observed] != 0]
    return dict(n=len(used), rmse=math.sqrt(np.mean(np.square(errors))) if errors else None,
                bias=float(np.mean(errors)) if errors else None,
                aard_percent=100 * float(np.mean(relative)) if relative else None,
                exceeds_2p2_percent=[r['record_id'] for r in used if abs(r[predicted] - r[observed]) > .022 * abs(r[observed])])


def integral_unavailability(rows, prefix):
    predictions = []
    for row in rows:
        if row[prefix+'_available']:
            continue
        reason = row[prefix+'_reason']
        kind = ('undefined' if reason == 'undefined_first_truncated_integral' else
                'outside_domain' if reason in ('above_ceiling', 'bubble_pressure_exceeded') else
                'not_evaluated' if reason.startswith('not_evaluated_after_interval:') else 'numerical_failure')
        predictions.append(dict(identity=row.get('record_id', f"{row['T_K']:g}K/loading={row['loading']:.12g}"),
                                reason=reason, kind=kind))
    return dict(count=len(predictions), counts={kind: sum(r['kind']==kind for r in predictions)
                for kind in ('undefined', 'outside_domain', 'not_evaluated', 'numerical_failure')}, predictions=predictions)


def crossings(x, y):
    return [dict(bracket=[a, b], direction='down' if u > v else 'up',
                 linear_zero=float(a - u * (b - a) / (v - u)))
            for a, b, u, v in zip(x, x[1:], y, y[1:]) if u * v < 0]


def pairs(rows, admitted):
    output, values = [], []
    for cold_run in (1, 2):
        for warm_run in (1, 2):
            parts = [sorted([r for r in rows if r['T_K'] == T and r['run'] == run
                             and (not admitted or r['interval_status'] == 'available')], key=lambda r: r['loading'])
                     for T, run in ((313.15, cold_run), (353.15, warm_run))]
            lo = max(p[0]['loading'] for p in parts)
            hi = min(p[-1]['loading'] for p in parts)
            x = sorted({r['loading'] for p in parts for r in p if lo <= r['loading'] <= hi})
            series = {}
            for field in ('observed', 'predicted') if admitted else ('observed',):
                y = [np.interp(x, [r['loading'] for r in p], [r[field] for r in p]) for p in parts]
                series[field] = y[1] - y[0]
                if field == 'observed':
                    scale = .022 * (np.abs(y[0]) + np.abs(y[1]))
            observed_crossings = crossings(x, series['observed'])
            for crossing in observed_crossings:
                indices = [x.index(a) for a in crossing['bracket']]
                crossing['uncertainty_resolved'] = all(abs(series['observed'][i]) > scale[i] for i in indices)
            model_crossings = crossings(x, series['predicted']) if admitted else []
            match = (len(model_crossings) == len(observed_crossings) and all(
                a['direction'] == b['direction'] and max(a['bracket'][0], b['bracket'][0]) <= min(a['bracket'][1], b['bracket'][1])
                for a, b in zip(observed_crossings, model_crossings)))
            output.append(dict(cold_run=cold_run, warm_run=warm_run, range=[lo, hi],
                               observed_crossings=observed_crossings, model_crossings=model_crossings,
                               outcome=('same nominal reversal pattern' if match else 'contradicts admitted nominal interval ordering')
                               if admitted and observed_crossings else 'no admitted experimental reversal to test' if admitted else 'unassessed full-table context',
                               uncertainty_resolved_reversal=any(r['uncertainty_resolved'] for r in observed_crossings)))
            values.extend(dict(cold_run=cold_run, warm_run=warm_run, loading=a, observed_difference=float(series['observed'][i]),
                               model_difference=float(series['predicted'][i]) if admitted else None,
                               uncertainty_bound=float(scale[i]), resolved_sign=1 if series['observed'][i] > scale[i] else -1 if series['observed'][i] < -scale[i] else 0)
                          for i, a in enumerate(x))
    return output, values


def state(T, a, w=.3, P=PRESSURE):
    key = (T, round(a, 12), w, P)
    if key in CACHE:
        return CACHE[key]
    row = dict(T_K=T, loading=a, mea_mass_fraction=w, P_Pa=P, status='numerical_failure')
    CACHE[key] = row
    try:
        if not 0 <= a <= CEILING:
            row['status'] = 'above_ceiling'
            return row
        # Bubble helper uses source molar masses; match the heat helper's exact feed.
        masses = {r['component_id']: r['fixed']['molar_mass']['value']['magnitude'] for r in c.MAPPING['components']}
        source_masses = c.TEMPLATE['reaction_system']['molar_masses_kg_per_mol']
        water = masses['monoethanolamine'] * (1-w) / (w*masses['water'])
        bubble_w = source_masses[1] / (source_masses[1] + water*source_masses[2])
        row['bubble_pressure_Pa'] = c.bubble_pressure(T, a, bubble_w)
        if row['bubble_pressure_Pa'] > P:
            row['status'] = 'bubble_pressure_exceeded'
            return row
        item = c.fo.heat_items(c.MAPPING, c.PARAMS, [dict(temperature_K=T, system_pressure_Pa=P,
            alpha_final=max(a, .01), observed_addition_enthalpy_J_per_mol_CO2=0., uH_J_per_mol_CO2=1000.)], c.HEAT_REQUEST)[0]
        problem = item.final_problem if a else item.problem
        result, checks = c.solved(problem)
        if max(checks[k] for k in c.IMBALANCES) > 1e-10 or not checks['requested_tolerance_met']:
            raise RuntimeError(f'equilibrium/balance criterion: {checks}')
        amounts = [result.amounts[0] * x for x in result.mole_fractions[:9]]
        row.update(checks, status='available', amounts_mol=amounts, mole_fractions=list(result.mole_fractions[:9]))
        PROBLEMS[key] = problem
    except (RuntimeError, ValueError) as error:
        row['cause'] = str(error)
    return row


def heat(T, a, b):
    endpoints = [state(T, v) for v in (a, b)]
    unavailable = [r['status'] for r in endpoints if r['status'] != 'available']
    if unavailable:
        return None, unavailable[0]
    initial, final = [PROBLEMS[(T, round(v, 12), .3, PRESSURE)] for v in (a, b)]
    item = c.fo.heat_items(c.MAPPING, c.PARAMS, [dict(temperature_K=T, system_pressure_Pa=PRESSURE,
        alpha_final=b, observed_addition_enthalpy_J_per_mol_CO2=0., uH_J_per_mol_CO2=1000.)], c.HEAT_REQUEST)[0]
    # heat_items already compiled the endpoints; preserve its feed/reference declaration.
    item.problem, item.final_problem, item.feed_amount = initial, final, b-a
    value = c.reg.predict(c.MODEL, item)
    if value.status.name != 'Available' or not value.requested_tolerance_met or not math.isfinite(value.value):
        return None, f'addition_enthalpy: {value.message}'
    return -value.value / 1000, 'available'


def differential(T, a, upper):
    previous, ladder = None, []
    for index in range(11):
        h = .001 / 2**index
        lo, hi = max(0., a - h / 2), min(upper, a + h / 2)
        q, reason = heat(T, lo, hi)
        ladder.append(dict(width=hi-lo, heat=q, status=reason))
        if q is None:
            break
        if previous is not None and abs(q - previous) <= HEAT_TOL:
            return q, dict(T_K=T, loading=a, passed=True, ladder=ladder)
        previous = q
    return None, dict(T_K=T, loading=a, passed=False, ladder=ladder)


def temperature_job(T):
    global CACHE, PROBLEMS
    CACHE, PROBLEMS = {}, {}
    c.s.RUNS = STAGING / f'runtime-{T}'
    source = [r for r in source_rows() if r['T_K'] == T]

    for a in grid:
        state(T, float(a))
    crossing = next(((a, b) for a, b in zip(grid, grid[1:]) if state(T, a)['status'] == 'available'
                     and state(T, b)['status'] == 'bubble_pressure_exceeded'), None)
    if crossing:
        lo, hi = crossing
        while hi-lo > .001:
            mid = (lo+hi)/2
            if state(T, mid)['status'] == 'available':
                lo = mid
            elif state(T, mid)['status'] == 'bubble_pressure_exceeded':
                hi = mid
            else:
                break
        crossing = [lo, hi]
    upper = crossing[0] if crossing else CEILING
    ladder, curves, fractions, comparisons = [], [], [], []
    for a in grid[1:]:
        row = state(T, float(a))
        curve = dict(T_K=T, loading=float(a), status=row['status'], integral_available=False, integral_reason=row['status'])
        if row['status'] == 'available':
            q, check = differential(T, float(a), upper)
            ladder.append(check)
            integral, reason = heat(T, 0., float(a))
            curve.update(differential=q, integral=integral, integral_available=integral is not None, integral_reason=reason)
            n = row['amounts_mol']
            carbon = sum(n[i] for i in (0, 4, 5, 6))
            f = dict(T_K=T, loading=float(a), carbamate=n[4]/carbon, bicarbonate=(n[5]+n[6])/carbon, free=n[0]/carbon)
            assert abs(sum(f[k] for k in ('carbamate', 'bicarbonate', 'free'))-1) <= 1e-10
            fractions.append(f)
        curves.append(curve)
    for r in source:
        a, prev = r['loading'], r['previous_loading']
        row = dict(r, observed=float(r['dHdiff_kJ_per_mol_CO2_reported']), predicted=None, interval_status='above_ceiling')
        if a <= CEILING:
            row['predicted'], row['interval_status'] = heat(T, prev, a)
        row['integral_co2_predicted'], row['integral_mea_predicted'], row['differential_predicted'] = None, None, None
        start = (0.051 if r['run'] == 1 else .036) if T == 393.15 else 0.
        reason = row['interval_status'] if row['interval_status'] in ('above_ceiling', 'bubble_pressure_exceeded') else 'not_evaluated_after_interval: '+row['interval_status']
        row.update(integral_co2_available=False, integral_mea_available=False, integral_co2_reason=reason, integral_mea_reason=reason)
        if T == 393.15 and a == start:
            row['integral_co2_reason'] = 'undefined_first_truncated_integral'
        if row['interval_status'] == 'available':
            q, check = differential(T, a, upper)
            row['differential_predicted'] = q
            ladder.append(check)
            integral, reason = heat(T, 0., a)
            row['integral_mea_predicted'] = None if integral is None else a*integral
            row.update(integral_mea_available=integral is not None, integral_mea_reason=reason)
            if a > start:
                row['integral_co2_predicted'], reason = heat(T, start, a)
                row.update(integral_co2_available=row['integral_co2_predicted'] is not None, integral_co2_reason=reason)
            row['scaled_residual'] = (row['predicted']-row['observed'])/(.022*abs(row['observed']))
        comparisons.append(row)
    print(f'{T:g} K: {len(comparisons)} source rows, {len(fractions)} admitted pathway states, {len(CACHE)} solves', flush=True)
    return dict(T_K=T, comparisons=comparisons, curves=curves, fractions=fractions,
                states=list(CACHE.values()), differential_checks=ladder, pressure_bracket=crossing)

def bottinger():
    output, checks = [], []
    packet = {o['identity']: o for o in c.PACKET}
    membership = read(ROOT / 'data/reference/MEA/manifests/speciation_target_membership.csv')
    canonical = {r['record_id']: r for r in read(c.s.CANONICAL_SPECIATION)}
    retained = read(EVIDENCE / 'species-figure-data.csv')
    for number in (*range(42, 51), *range(55, 60), *range(63, 68)):
        identity = f'Bottinger2008_state_{number:03d}'
        observation = packet[identity]
        request = c.s.corrected_request(observation['request'], c.s.reaction_values(c.MAPPING), mapping=c.MAPPING)
        problem = c.s._problem_from_request(request)
        result, diagnostics = c.solved(problem, feed=request['reaction_system']['analytical_feed_contract']['neutral_analytical_amounts_mol'])
        assert max(diagnostics[k] for k in c.IMBALANCES) <= 1e-10 and diagnostics['requested_tolerance_met']
        x = result.mole_fractions[:9]
        rows = [r for r in membership if r['membership_id'].startswith(identity+'|')]
        raw = {r['species']: float(r['value_mole_fraction']) for m in rows if m['measurement_identity']
               and (r := canonical[m['measurement_identity']])}
        # Same-state carbon-pool sum; do not divide mole fractions by analytical loading.
        source_loading = float(rows[0]['co2_loading_mol_per_mol_mea'])
        carbon_obs = raw['CO2'] + raw['MEACOO-'] + raw['HCO3-']
        carbon_model = sum(x[i] for i in (0, 4, 5, 6))
        source_w = float(rows[0]['mea_mass_fraction'])
        for species, pred, obs in (('carbamate', x[4]/carbon_model, raw['MEACOO-']/carbon_obs),
                                   ('bicarbonate', (x[5]+x[6])/carbon_model, raw.get('HCO3-', 0.)/carbon_obs)):
            output.append(dict(identity=identity, T_K=float(problem.T), P_Pa=float(problem.P), loading=source_loading,
                               mea_mass_fraction=source_w, species=species, predicted=pred, observed=obs,
                               basis='total liquid CO2; same-state source carbon-pool sum'))
        for old in (r for r in retained if r['identity'] == identity):
            selector = {'MEA + MEAH+': (1, 3), 'MEACOO-': (4,), 'HCO3-': (5, 6)}[old['species']]
            new = sum(x[i] for i in selector)
            error = abs(new-float(old['predicted']))/max(abs(new), abs(float(old['predicted'])), 1e-300)
            checks.append(dict(identity=identity, species=old['species'], relative_error=error, passed=error <= 1e-6))
    return output, checks

            value = difference(mid)
            if value is None:

            if value*left > 0:
                lo, left = mid, value
            else:
                hi = mid
        ends = [difference(lo), difference(hi)]
        if all(v is not None for v in ends) and min(map(abs, ends)) <= 2*HEAT_TOL:
            # Avoid certifying a sign at an endpoint indistinguishable from zero.
            center = lo-ends[0]*(hi-lo)/(ends[1]-ends[0])
            lo, hi = max(crossing['bracket'][0], center-.00045), min(crossing['bracket'][1], center+.00045)
            ends = [difference(lo), difference(hi)]
        assert hi-lo <= .001 and all(v is not None and abs(v)>2*HEAT_TOL for v in ends) and ends[0]*ends[1]<0
        crossing['refined_bracket'] = [lo, hi]
        crossing['endpoint_differences_kJ_mol'] = ends
        crossing['resolved_on_numerical_heat_tolerance'] = all(v is not None and abs(v)>2*HEAT_TOL for v in ends)
