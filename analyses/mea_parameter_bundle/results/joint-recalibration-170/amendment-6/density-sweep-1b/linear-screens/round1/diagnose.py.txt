# Linear least squares with Gx <= h by feasible active-set steps. Every objective step uses SVD, not normal equations.
def qpls(A,b,G,h,x=None,maxiter=1500):
 n=A.shape[1]; x=np.zeros(n) if x is None else x.copy(); G=np.array(G,float); h=np.array(h,float)
 norms=np.linalg.norm(G,axis=1); valid=norms>1e-14; G=G[valid]/norms[valid,None];h=h[valid]/norms[valid]
 if np.max(G@x-h)>1e-7:raise ValueError(('not feasible',np.max(G@x-h)))
 active=[]
 # Initially active independent constraint rows only.
 for i in np.flatnonzero(abs(G@x-h)<1e-9):
  if np.linalg.matrix_rank(G[active+[int(i)]],tol=1e-10)>len(active):active.append(int(i))
 for it in range(maxiter):
  C=G[active] if active else np.empty((0,n)); Z=np.linalg.svd(C,full_matrices=True)[2][len(active):].T if active else np.eye(n)
  y=np.linalg.lstsq(A@Z,-(A@x+b),rcond=1e-12)[0] if Z.shape[1] else np.empty(0); p=Z@y
  if np.linalg.norm(p)<1e-8*(1+np.linalg.norm(x)):
   g=A.T@(A@x+b); mu=np.linalg.lstsq(C.T,-g,rcond=1e-12)[0] if active else np.empty(0)
   if not active or min(mu)>=-1e-8:
    return x,{'iterations':it,'max_violation':float(max(G@x-h)),'stationarity':float(np.max(abs(g+C.T@mu))) if active else float(np.max(abs(g))),'minimum_multiplier':float(min(mu)) if active else 0.}
   del active[int(np.argmin(mu))];continue
  gp=G@p; slack=h-G@x; candidates=np.flatnonzero(gp>1e-10); candidates=np.array([i for i in candidates if i not in active],int)
  alpha=1.;block=None
  if len(candidates):
   ratios=slack[candidates]/gp[candidates]; k=np.argmin(ratios)
   if ratios[k]<1-1e-10:alpha=max(0.,float(ratios[k]));block=int(candidates[k])
  x+=alpha*p
  if block is not None:
   if np.linalg.matrix_rank(G[active+[block]],tol=1e-10)>len(active):active.append(block)
   else:raise RuntimeError('dependent blocker')
 raise RuntimeError(('active set iterations',np.linalg.norm(p),np.max(G@x-h)))
