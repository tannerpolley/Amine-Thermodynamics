def eigen_pivots(J):
    """Eigen FullPivLU: complete pivoting; rank counts |pivot| > |max pivot| * min(m,n) * eps."""
    U = J.copy(); m, n = U.shape; piv = []
    rows, cols = list(range(m)), list(range(n))
    for k in range(min(m, n)):
        sub = np.abs(U[k:, k:]); i, j = np.unravel_index(np.argmax(sub), sub.shape); i += k; j += k
        U[[k, i]] = U[[i, k]]; U[:, [k, j]] = U[:, [j, k]]; rows[k], rows[i] = rows[i], rows[k]; cols[k], cols[j] = cols[j], cols[k]
        p = U[k, k]; piv.append(float(p))
        if p != 0: U[k+1:, k:] -= np.outer(U[k+1:, k] / p, U[k, k:])
    thr = abs(piv[0]) * min(m, n) * np.finfo(float).eps
    return piv, thr, sum(abs(p) > thr for p in piv), rows, cols
