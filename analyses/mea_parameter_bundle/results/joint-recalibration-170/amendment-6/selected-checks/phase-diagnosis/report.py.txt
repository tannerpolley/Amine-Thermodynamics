def table(path,rows):
    with path.open('w',newline='') as h:
        w=csv.DictWriter(h,fieldnames=list(dict.fromkeys(k for r in rows for k in r)))
        w.writeheader();w.writerows(rows)

desc=[]; trials=[]; points=[]
for index in range(73):
    p=original/f'point-{index}.json';d=load(p);points.append(d)
    for ri,r in enumerate(d['stability']):
        for ti,t in enumerate(r['trials']):
            tr=dict(index=index,trial_index=ti,source=str(p),phase_kind=t['phase']['kind'],seed=t['seed'],
                evidence=t['evidence'],optimizer_success=t['optimizer']['success'],message=t['message'],
                tpd_over_rt=t['quantities'].get('tpd_over_rt'),
                descent_error_estimate_over_rt=t['quantities'].get('descent_error_estimate_over_rt'))
            trials.append(tr)
            if not t['resolved_descent']:continue
            donor=d['final']['phases'][t['donor']]
            native=roots[(index,ti)]
            row=dict(index=index,trial_index=ti,temperature_K=d['temperature_K'],loading=d['loading'],
                pressure_Pa=d['pressure_Pa'],phase_kind=t['phase']['kind'],seed=t['seed'],donor=t['donor'],
                molar_density_mol_m3=1/t['quantities']['volume_m3_per_mol'],
                density_branch=native['liquid']['density_branch'],stable_root_count=native['liquid']['stable_root_count'],
                pressure_density_derivative=native['liquid']['pressure_density_derivative'],
                enriched_species=';'.join(species[i] for i,x in enumerate(t['composition']) if x>donor['mole_fractions'][i]),
                **{k:t['quantities'][k] for k in ['tpd_over_rt','descent_error_estimate_over_rt','donor_slope_over_rt',
                    'split_delta_g_over_rt','split_error_estimate_over_rt','split_beta_mol']},
                **{'x_'+s:x for s,x in zip(species,t['composition'],strict=True)},
                **{'donor_x_'+s:x for s,x in zip(species,donor['mole_fractions'],strict=True)})
            desc.append(row)
