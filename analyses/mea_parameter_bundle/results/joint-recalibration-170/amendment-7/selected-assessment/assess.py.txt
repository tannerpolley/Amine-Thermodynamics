def safe(task):
    kind, row = task
    try:
        value = {'density':density, 'cp':cp, 'heat':heat, 'state':state, 'carbonate':carbonate, 'declared-density':declared_density}[kind](row)

def density(row):
    identity = row['id']; a = float(row.get('co2_loading_mol_per_mol_mea') or 0)
    T = float(row['temperature_C'])+273.15; w = float(row['mea_mass_fraction'])
    P = float(row['pressure_kPa'])*1000 if row.get('pressure_kPa') else 100000.
    obs = float(row['value'])*(1000 if row['value_unit']=='g/cm^3' else 1)
    if w==1 and a==0:
        model = epcsaft.Pure(epcsaft.Parameters.from_mapping(MAPPING, components=['monoethanolamine']))
        state = model.state(T, P=P, phase='liquid')
        mass = next(c['fixed']['molar_mass']['value']['magnitude'] for c in MAPPING['components'] if c['component_id']=='monoethanolamine')
        rho = state.molar_density*mass
        evidence = dict(route='pure MEA native state', dP_drho=state.pressure_density_derivative)
    else:
        if a==0:
            problem = s.unloaded_density_problem(MAPPING,TEMPLATE,dict(T_K=T,P_Pa=P,MEA_weight_fraction=w))
        else:
            req = he.request(MAPPING,T,a,density=True)
            req = s.source_feed_request(req,dict(source_key='Amundsen2009', CO2_loading=a,temperature_reported_C=T-273.15,MEA_weight_fraction=w))
            problem = s._problem_from_request(s.corrected_request(req,s.reaction_values(MAPPING),mapping=MAPPING))
            problem.P = P
        model = epcsaft.Mixture(PARAMS, thermochemistry=he.vrc.record())
        item = reg.observation(PARAMS,'phase_density',problem,observed=obs,form='difference',scale=1.,phase=problem.phases[0].name,mass_basis=True)
        pred = reg.predict(model,item)
        if pred.status.name!='Available': raise RuntimeError(f'{pred.status.name}: {pred.message}')
        rho = pred.value
        solved = eq.solve_equilibrium(model,problem)
        if not solved.success: raise RuntimeError(solved.message)
        point = eq.compile_problem(model,problem).evaluate(solved.coordinates)
        evidence = dict(route='unloaded_density_problem' if a==0 else 'source_feed_request and native phase_density',max_abs_raw_residual=max(map(abs,point.values)),requested_tolerance_met=pred.requested_tolerance_met,stable_root_count=pred.stable_root_count,
            pressure_Pa=solved.pressure,temperature_K=solved.temperature,reduced_hessian_minimum=solved.reduced_hessian_minimum)
    scale = SCALES.get(identity,{}).get('scale')
    if scale is None and a==0 and row['source_key']=='Amundsen2009': scale=.5
    error = rho-obs
    return dict(id=identity,source=row['source_key'],T_K=T,P_Pa=P,loading=a,mea_mass_fraction=w,observed_kg_m3=obs,predicted_kg_m3=rho,error_kg_m3=error,error_percent=100*error/obs,s_kg_m3=scale,error_over_s=error/scale if scale else None,
        source_uncertainty_value=row.get('uncertainty_value'),source_uncertainty_unit=row.get('uncertainty_unit'),source_expanded_uncertainty_value=row.get('density_expanded_uncertainty_value'),source_expanded_uncertainty_unit=row.get('density_expanded_uncertainty_unit'),source_uncertainty_scope=row.get('uncertainty_scope'),source_locator=row['source_locator'],lifecycle_status=row.get('lifecycle_status'),numerical_evidence=evidence,parameter_sha256=SHA)

def cp(row):
    proxy=row['proxy']; T=row['T_K']; a=row['source_loading']
    value=he.cp_check(MAPPING,T,a,row['source_observed_kJ_kg_K'],row=row,proxy_loading=proxy)
    feed=dict(row['analytical_feed'],**{'carbon-dioxide':max(a,proxy)})
    declared_masses={c['component_id']:c['fixed']['molar_mass']['value']['magnitude'] for c in MAPPING['components']}
    declared_mass=sum(n*declared_masses[c] for c,n in feed.items())
    helper_mass=sum(feed.get(c,0)*m for c,m in zip(s.COMPONENT_IDS,he.hc.MOLAR_MASS))/1000
    # The helper's hard-coded mass vector differs from the declared frozen masses.
    # Re-express its native extensive Cp using the design's complete declared feed mass.
    value['helper_mass_basis_values']=dict(value,helper_mass_kg=helper_mass)
    for key in ('equilibrium_cp_kJ_kg_K','frozen_cp_kJ_kg_K'):value[key]*=helper_mass/declared_mass
    value.update(declared_feed_mass_kg=declared_mass,declared_molar_masses_kg_mol=declared_masses,analytical_feed=feed,deviation_percent=100*(value['equilibrium_cp_kJ_kg_K']/row['source_observed_kJ_kg_K']-1))
    model=epcsaft.Mixture(PARAMS,thermochemistry=he.vrc.record())
    def solved_at(t):
        problem=s._problem_from_request(he.request(MAPPING,t,max(a,1e-5)))
        problem.feed=eq.Amounts(dict(row['analytical_feed'],**{'carbon-dioxide':max(a,proxy)}))
        problem.P=row['P_Pa']
        result=eq.solve_equilibrium(model,problem)
        if not result.success or max(map(abs,result.residuals))>1e-9: raise RuntimeError(result.message)
        (h,_),(n,_)=he.vrc.first_actions(model,problem,result,[he.vrc.observable('PhaseProperty',0,prop=epcsaft.PropertyObservable.TotalEnthalpy),he.vrc.observable('PhaseAmount',0)],[(0.,0.,None)])[0]
        mass=sum(n*declared_masses[c] for c,n in problem.feed.values.items())
        return dict(T_K=t,H_J=h*n,mass_kg=mass,max_abs_raw_residual=max(map(abs,result.residuals)),iterations=result.iterations if hasattr(result,'iterations') else None)
    ladder=[]
    for dt in (.1,.05):
        lo,hi=solved_at(T-dt),solved_at(T+dt)
        fd=(hi['H_J']-lo['H_J'])/(2*dt)/lo['mass_kg']/1000
        ladder.append(dict(step_K=dt,minus=lo,plus=hi,cp_kJ_kg_K=fd,relative_error=abs(fd-value['equilibrium_cp_kJ_kg_K'])/abs(fd)))
    coarse,fine=ladder
    value.update(id=row['id'],proxy=proxy,enthalpy_ladder=ladder,resolve_passed=fine['relative_error']<=1e-6 and abs(fine['cp_kJ_kg_K']-value['equilibrium_cp_kJ_kg_K'])<=abs(coarse['cp_kJ_kg_K']-value['equilibrium_cp_kJ_kg_K'])/2,source_locator=row['source_locator'],parameter_sha256=SHA)
    return value

def tasks_for(mode):
    tasks=[]
    if mode=='density':
        for name in ('Amundsen_2009_density_viscosity','Hartono_2014_loaded_density','Han_2012_loaded_density'):
            path=Path('data/reference/MEA/observations/density_viscosity')/(name+'.csv')
            rows=[r for r in csv.DictReader(path.open()) if r['property']=='density']
            for i,row in enumerate(rows,1):
                # Full source-table order provides the historical Amundsen state IDs.
                row['id']=row.get('record_id') or f'Amundsen2009_state_{i:03d}'
                tasks.append(('density',row))
        write('density-inputs.json',tasks)
    if mode=='cp':
        for row in SPEC['direct_rows']:
            if row['quantity']=='equilibrium_isobaric_heat_capacity':
                tasks.extend(('cp',dict(row,proxy=p)) for p in ((1e-6,1e-7) if row['source_loading']==0 else (1e-6,)))
    if mode=='calorics':tasks.extend(('heat',r) for r in fo.load(SPEC['heat_rows_path']))
    if mode=='states':
        groups=json.loads((OUT/'observation-groups.json').read_text())
        fe.binding('Held',str(RECORD))
        for kind,rows in groups.items():
            for o in rows:
                T=o['request']['temperature']['value']
                reason=fe.probe.reaction_domain_status(o)
                if reason or not 293.15<=T<=393.15:
                    write('excluded/'+kind+'-'+o['identity'].replace(':','_')+'.json',dict(group=kind,observation=o,status='Inadmissible',reason=reason or 'outside 293.15–393.15 K'))
                    continue
                tasks.append(('state',dict(group=kind,observation=o)))
        write('state-inputs.json',tasks)
    if mode=='declared-density':
        for p in sorted((OUT/'density-rows').glob('*.json')):
            r=json.loads(p.read_text())
            if r['input']['id'] in DIRECT_ITEMS:tasks.append(('declared-density',dict(id=r['input']['id'],retained_row=str(p.relative_to(OUT)))))
    if mode=='carbonate':
        source=Path('data/reference/MEA/observations/liquid_speciation/Jakobsen_2005_ChEq.csv')
        for i,row in enumerate(csv.DictReader(source.open()),1):
            if float(row['MEA_weight_fraction'])==.3 and float(row['maxload'])<=.5:
                tasks.append(('carbonate',dict(row,id=f'Jakobsen2005_state_{i:03d}')))
    return tasks


def main():
    mode=sys.argv[1]
    inventory()
    if mode=='inventory':return
    global DIRECT_ITEMS
    if mode=='declared-density':
        _,rows=fo.stage6_rows(SPEC,MAPPING)
        DIRECT_ITEMS={r['id']:(r,item) for r,item in rows if r['quantity']=='phase_density'}
    tasks=tasks_for(mode)
    # Independent fixed states use every available core, with BLAS/OpenMP at one thread.
    workers = min(os.cpu_count(), max(1, int(next(x for x in open('/proc/meminfo') if x.startswith('MemAvailable')).split()[1]) // 1_000_000))
    with cf.ProcessPoolExecutor(max_workers=workers) as pool:
        futures={pool.submit(safe,t):i for i,t in enumerate(tasks)}
        for f in cf.as_completed(futures):
            i=futures[f];result=f.result()
            write(f'{mode}-rows/{i:04d}.json',result)
            print(mode,i,result['status'],result.get('error',''),flush=True)
    write(mode+'-complete.json',dict(tasks=len(tasks)))


if __name__=='__main__': main()
