from diagnose import *
def solve_hard(d,s,tol,metric='c149'):
 x0,v,pk=feasible(s,d,tol)
 if v>1e-7:return None,{'feasible':False,'slack':v,'kkt':pk}
 mask=d[metric];G,h=constraints(s,d,tol);x,k=qpls(s['J'][mask]*np.sqrt(d['w'][mask,None]),d['r'][mask]*np.sqrt(d['w'][mask]),G,h,x0);return x,k

def minimax_cap(d,s,cap):
 lower=0.;upper=metrics(d,d['r'])['density_max'];best=None
 # The unchanged endpoint is feasible in the box and under the cost cap.
 for _ in range(32):
  mid=(lower+upper)/2;x,k=solve_hard(d,s,mid)
  if x is not None and metrics(d,d['r']+s['J']@x)['C149']<=cap:
   upper=mid;best=report(d,s,x,k)
  else:lower=mid
 return {'C149_cap':cap,'density_max_lower':lower,'density_max_upper':upper,'result':best}
