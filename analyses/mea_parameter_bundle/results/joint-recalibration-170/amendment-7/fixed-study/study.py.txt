"""Disclosed fixed-vector probe for #170 amendment 7 steps 1-3; no fitting.

A vector is the Held A4 control (design path.py.txt mapping: all six DH=sigma,
absent MEA-water slope = 0) moved toward the selected A6 vector by one path
fraction per coordinate block (A7-1 applied block by block). Every changed
identity belongs to exactly one of the six design blocks (A7 §5 step 2); both
coefficients of each reaction law move together.

    study.py.txt family LABEL SPEC   all 73 family points
    study.py.txt points LABEL SPEC   points 63/70/72
    study.py.txt exchanges           the 12 step-2 exchanges, points 63/70/72
    study.py.txt subexchanges        the 8 single ion-water exchanges, points 63/70/72
    study.py.txt splits LABEL        one split-seeded solve per descending point
    study.py.txt score LABEL         exact 307-row replay and A6 gates
    study.py.txt screen LABEL PATH   all 73 family points for a parameter document

SPEC is "default" or "default,name=fraction,...", where name is a block or a
single ion-water coefficient, e.g. "0.125,meah-size=0.25".
Run from the worktree root through the pinned interpreter, GNU time and nice.
"""
import copy
import hashlib
import json
import math
from pathlib import Path
import sys

import final_evidence as f
from heat_evidence_170 import request

root = Path('analyses/mea_parameter_bundle/results/joint-recalibration-170')
output = root/'amendment-7/fixed-study'
a4_path = root/'amendment-4/matched-larger-dV/control/usable-budget-endpoint.json'
a6_path = root/'amendment-6/joint-mea-water/held/fit/usable-budget-endpoint.json'
assert hashlib.sha256(a4_path.read_bytes()).hexdigest() == '365758b80c7f701ac7c09c4319d4e216cf5b270374e856381f7d2ba8d1ea7f29'
assert hashlib.sha256(a6_path.read_bytes()).hexdigest() == 'ceefc95cd051fb291e0c08675c4cb6f159bf10dca0116ef1251ba78c243f79c7'
a4 = f.s.parameter_mapping(a4_path)
a6 = f.s.parameter_mapping(a6_path)
old = {n['identity']: n for n in f.s._identified(a4)}
new = {n['identity']: n for n in f.s._identified(a6)}
slope = 'pair/monoethanolamine/water/k_ij/reciprocal_temperature_slope'
assert new.keys() - old.keys() == {slope} and not old.keys() - new.keys()
for identity, node in old.items():
    if identity.endswith('/debye_huckel_diameter'):
        node['value']['magnitude'] = old[identity.replace('/debye_huckel_diameter', '/segment_diameter')]['value']['magnitude']
start = {k: 0.0 if k == slope else old[k]['value']['magnitude'] for k in new}
finish = {k: n['value']['magnitude'] for k, n in new.items()}
meah, carb = 'component/protonated-monoethanolamine/', 'component/carbamate-anion/'
BLOCKS = {
    'meah-size': [meah+'segment_diameter', meah+'packing_diameter', meah+'debye_huckel_diameter'],
    'ion-energies': [meah+'dispersion_energy_over_k', carb+'dispersion_energy_over_k'],
    'ion-water': ['pair/bicarbonate-anion/water/k_ij', 'pair/carbamate-anion/water/k_ij',
                  'pair/protonated-monoethanolamine/water/k_ij', 'pair/protonated-monoethanolamine/water/k_ij/reciprocal_temperature_slope'],
    'mea-water': ['pair/monoethanolamine/water/k_ij', slope],
    'r2': ['reaction:R2:correlation:a', 'reaction:R2:correlation:b_k'],
    'r5': ['reaction:R5:correlation:a_k', 'reaction:R5:correlation:b'],
}
changed = {k for k in new if start[k] != finish[k]}
assert sorted(sum(BLOCKS.values(), [])) == sorted(changed), changed ^ set(sum(BLOCKS.values(), []))
POINTS = [63, 70, 72]
f.s.verify_wheel()


# Single ion-water coefficients, for the owner-approved sub-block probe (#170 step-4 decision).
SINGLE = {'carbamate-water': 'pair/carbamate-anion/water/k_ij', 'meah-water': 'pair/protonated-monoethanolamine/water/k_ij',
          'bicarbonate-water': 'pair/bicarbonate-anion/water/k_ij', 'meah-water-slope': 'pair/protonated-monoethanolamine/water/k_ij/reciprocal_temperature_slope'}
assert sorted(SINGLE.values()) == sorted(BLOCKS['ion-water'])


def fractions(spec):
    head, *rest = spec.split(',')
    out = dict.fromkeys(changed, float(head))
    for item in rest:
        name, value = item.split('=')
        for k in BLOCKS.get(name) or [SINGLE[name]]:
            out[k] = float(value)
    assert all(0 <= v <= 1 for v in out.values())
    return out


def vector(share):
    data = copy.deepcopy(a6)
    for node in f.s._identified(data):
        k = node['identity']
        if k in share:
            node['value']['magnitude'] = (1-share[k])*start[k] + share[k]*finish[k]
    return data


# Uniform fractions must reproduce the design's path vectors exactly.
for t in ('0p125', '0p25'):
    assert vector(fractions(t.replace('p', '.'))) == f.s.parameter_mapping(root/f'amendment-7/design-probes/path-{t}/fixed-parameters.json')


def build(label, spec):
    share = fractions(spec)
    data = vector(share)
    run = output/label
    run.mkdir(parents=True, exist_ok=False)
    f.s.write_json(run/'fixed-parameters.json', data)
    f.s.write_json(run/'provenance.json', dict(spec=spec, block_fractions=share, blocks=BLOCKS,
        values={k: (start[k], finish[k], n['value']['magnitude']) for n in f.s._identified(data) for k in [n['identity']] if k in changed},
        a4_path=str(a4_path), a4_sha256=f.s.sha256(a4_path), a6_path=str(a6_path), a6_sha256=f.s.sha256(a6_path),
        wheel_path=str(f.s.installed_wheel()), wheel_sha256=f.s.ENGINE_WHEEL_SHA256, module=f.epcsaft.__file__,
        script_sha256=f.s.sha256(Path(__file__)), form='Held: all six DH=sigma; packing unchanged at 0.88 sigma'))
    return data, run


def reduce(run):
    summary = []
    for path in sorted(run.glob('point-*.json'), key=lambda p: int(p.stem.split('-')[1])):
        row = json.loads(path.read_text())
        trials = [t for r in row.get('stability', []) for t in r['trials']]
        descents = [t for t in trials if t['resolved_descent']]
        summary.append(dict(index=row['index'], classification=row['classification'], final_success=row.get('final', {}).get('success', False),
            resolved_trials=len(descents), minimum_resolved_tpd_over_rt=min((t['quantities']['tpd_over_rt'] for t in descents), default=None),
            minimum_split_delta_g_over_rt=min((t['quantities']['split_delta_g_over_rt'] for t in descents), default=None),
            failed_trial_optimizers=sum(not t['optimizer']['success'] for t in trials),
            evidence=sorted({t['evidence'] for t in trials})))
    f.s.write_json(run/'reduction.json', summary)
    print(run.name, 'points', len(summary), 'descent_points', [r['index'] for r in summary if r['resolved_trials']],
          'min_tpd', min((r['minimum_resolved_tpd_over_rt'] for r in summary if r['resolved_trials']), default=None),
          'classifications', {k: sum(r['classification'] == k for r in summary) for k in ('failed', 'unresolved', 'passed-two-phase', 'passed-single-liquid')}, flush=True)


def points(data, run, indices):
    for index in indices:
        prior = json.loads((root/f'amendment-6/selected-checks/phase-family/point-{index}.json').read_text())
        row = {k: prior[k] for k in ('index', 'loading', 'temperature_K', 'pressure_Pa')}
        f.family_job(data, request(data, 313.15, .11), row, [], run/f'point-{index}.json')
    reduce(run)


def splits(label):
    run = output/label
    mapping = f.s.parameter_mapping(run/'fixed-parameters.json')
    model = f.epcsaft.Mixture(f.epcsaft.Parameters.from_mapping(mapping), thermochemistry=f.thermal.record())
    summary = []
    for path in sorted(run.glob('point-*.json'), key=lambda p: int(p.stem.split('-')[1])):
        old_row = json.loads(path.read_text())
        trials = [t for r in old_row.get('stability', []) for t in r['trials'] if t['resolved_descent']]
        if not trials:
            continue
        trial = min(trials, key=lambda t: t['quantities']['split_delta_g_over_rt'])
        phases = [f.eq.Phase(p['identity'], kind=p['role'], support=p['support']) for p in old_row['final']['phases']]
        phases.append(f.eq.Phase('L2', kind='liquid', support=f.s.COMPONENT_IDS))
        problem = f.flash_problem(request(mapping, 313.15, .11), old_row['loading'], old_row['temperature_K'], old_row['pressure_Pa'], phases)
        cursor, amounts, compositions, packings = 0, [], [], []
        for phase in phases:
            count = len(phase.support)
            compositions.append([math.exp(v) for v in trial['split_coordinates'][cursor:cursor+count]])
            amounts.append(trial['split_coordinates'][cursor+count])
            packings.append(math.exp(trial['split_coordinates'][cursor+count+1]))
            cursor += count+2
        assert cursor == len(trial['split_coordinates'])
        begin = f.eq.EquilibriumStart(compositions=compositions, amounts=amounts, packing_fractions=packings)
        batch = f.eq.solve_equilibrium_starts(model, problem, [begin], f.eq.EquilibriumWorkLimits(attempts=1, attempt_iterations=400, iterations=400))
        result = batch.results[0]
        row = dict(label=label, index=old_row['index'], seed=trial['seed'], selected=batch.selected, causes=list(batch.causes),
            original_G_over_RT=old_row['final']['objective'], finite_split_delta_G_over_RT=trial['quantities']['split_delta_g_over_rt'],
            finite_split_error_over_RT=trial['quantities']['split_error_estimate_over_rt'])
        if result is not None:
            row['result'] = f.phase_payload(model, problem, result)
            row['delta_G_over_RT'] = result.objective-old_row['final']['objective']
        f.s.write_json(run/f"split-seeded-{old_row['index']}.json", row)
        summary.append({k: v for k, v in row.items() if k != 'result'} | dict(success=result is not None and result.success,
            requested_tolerance_met=result is not None and result.requested_tolerance_met))
        print(summary[-1], flush=True)
    f.s.write_json(run/'split-summary.json', summary)


def score(label):
    import fit_campaign as c
    import fit_observations as o
    run = output/label
    record = run/'fixed-parameters.json'
    spec = c.load(root/'amendment-6/joint-mea-water/held/fit-spec.json')
    mapping = o.s.parameter_mapping(record)
    _, rows = o.stage6_rows(spec, mapping)
    weights = c.stage6_weights(spec, [r for r, _ in rows], 1.)
    values = o.replay(mapping, [(r['id'], item, w) for (r, item), w in zip(rows, weights, strict=True)])
    o.s.write_json(run/'full-replay.json', values)
    assert len(values) == 307 and all(v['status'] == 'Available' for v in values)
    kind = {r['id']: r['quantity'] for r, _ in rows}
    free = ('AMU-DENS-002', 'AMU-DENS-009', 'AMU-DENS-016', 'AMU-DENS-023', 'AMU-DENS-030')
    loaded = set(spec['loaded_density_ids'])
    part = lambda keep: math.fsum(v['cost'] for v in values if keep(v['target']))
    ard = lambda q: 100*math.fsum(abs(v['predicted']/v['observed']-1) for v in values if kind[v['target']] == q)/sum(kind[v['target']] == q for v in values)
    heat = [v['predicted']-v['observed'] for v in values if kind[v['target']] == 'AdditionEnthalpy']
    dens = [v for v in values if v['target'] in loaded]
    summary = dict(label=label, parameter_sha256=o.s.sha256(record), cost=part(lambda i: True),
        C149=part(lambda i: kind[i] in ('PartialPressure', 'PhaseMoleFraction', 'AdditionEnthalpy')),
        cai_cost=part(lambda i: i.startswith('cai1996')), loaded_density_cost=part(lambda i: i in loaded),
        pressure_AARD_pct=ard('PartialPressure'), species_AARD_pct=ard('PhaseMoleFraction'),
        vinjarapu_rmse_kJ_mol=math.sqrt(math.fsum(x*x for x in heat)/len(heat))/1000,
        loaded_within_2s=sum(abs(v['residual']) <= 2 for v in dens), loaded_count=len(dens),
        loaded_max_abs_error_over_s=max(abs(v['residual']) for v in dens),
        co2_free_errors={v['target']: v['predicted']-v['observed'] for v in values if v['target'] in free})
    o.s.write_json(run/'score.json', summary)
    print(summary, flush=True)


def family(data, run):
    f.phase_family(data, request(data, 313.15, .11), root/'amendment-6/inputs/revision-3/phase/family-points.txt',
        '6e1231d9506fd6dd9e965a64120ff952cebf1c4778da5d66cfb0f888e8b7f44b', run)
    reduce(run)


mode = sys.argv[1]
if mode in ('family', 'points'):
    data, run = build(sys.argv[2], sys.argv[3])
    family(data, run) if mode == 'family' else points(data, run, POINTS)
elif mode == 'screen':
    run = output/sys.argv[2]
    run.mkdir(parents=True, exist_ok=False)
    family(f.s.parameter_mapping(Path(sys.argv[3])), run)
elif mode == 'exchanges':
    for lo, hi in ((0.125, 0.25), (0.25, 0.125)):
        for block in BLOCKS:
            label = f"exchange-{str(lo).replace('.', 'p')}-{block}-to-{str(hi).replace('.', 'p')}"
            data, run = build(label, f'{lo},{block}={hi}')
            points(data, run, POINTS)
elif mode == 'subexchanges':
    for lo, hi in ((0.125, 0.25), (0.25, 0.125)):
        for name in SINGLE:
            label = f"exchange-{str(lo).replace('.', 'p')}-{name}-to-{str(hi).replace('.', 'p')}"
            data, run = build(label, f'{lo},{name}={hi}')
            points(data, run, POINTS)
elif mode == 'splits':
    splits(sys.argv[2])
elif mode == 'score':
    score(sys.argv[2])
else:
    raise ValueError(mode)
