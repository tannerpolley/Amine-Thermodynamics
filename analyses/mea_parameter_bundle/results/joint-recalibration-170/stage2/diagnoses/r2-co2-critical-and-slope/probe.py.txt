def mapping(k,eps=169.21,slope=None):

 if slope is not None:
  pair=next(v for v in m['pairs'] if {v['component_id_a'],v['component_id_b']}=={'carbon-dioxide','water'})
  c=copy.deepcopy(pair['coefficients'][0]);c.update(family='k_ij_reciprocal_temperature_slope',identity=case['coordinate']['identity']+'/reciprocal_temperature_slope',value={'magnitude':slope,'unit':'kelvin'},reference_temperature={'magnitude':313.15,'unit':'kelvin'});pair['coefficients'].append(c)

def fit(k,eps=169.21,slope=True):
 m=mapping(k,eps,0 if slope else None);p,rows=o.stage2_rows(spec,m,case)
 c=case['coordinate'];coords=[reg.coordinate(p,[('k_ij',tuple(c['target']),1.)],origin=k,scale=c['scale'],bounds=tuple(c['bounds']))]
 if slope:coords.append(reg.coordinate(p,[('k_ij_reciprocal_temperature_slope',tuple(c['target']),1.)],origin=0,scale=10,bounds=(-1000,1000)))
 f=reg.fit(p,coords,[r[1] for r in rows],weights=[r[0]['weight'] for r in rows],thermochemistry=o.thermal.record(),controls=reg.FitControls(**spec['controls'],workers=1,maximum_elapsed_time_seconds=50),singular_vectors=True)
 out={v:o.s._jsonable(getattr(f,v)) for v in ('status','message','usable','initial_cost','final_cost','physical','optimizer','predictions','residuals','weighted_residuals','optimizer_jacobian','physical_jacobian','iterations','iteration_costs','iteration_seconds','trial_failures','residual_evaluations','jacobian_evaluations')}
 out.update(start_k=k,eps=eps,coordinates=['k','b'] if slope else ['k'],targets=[r[0]['id'] for r in rows],statuses=[v.name for v in f.statuses],covariance={v:o.s._jsonable(getattr(f.covariance,v)) for v in ('status','rank','singular_values','active_bounds','projected_gradient_norm')},failures=[dict(trial=v.trial,observation=v.observation,status=v.status.name,message=v.message,jacobian=v.jacobian) for v in f.failures])

  scores.append(dict(T=T,n=len(ix),mean_absolute_percent=sum(map(abs,err))/len(ix),max_absolute_percent=max(map(abs,err)),signed_percent=err))
 out['isotherms']=scores;save(sys.argv[1]+'.json',out);print(json.dumps({k:out[k] for k in ('status','message','physical','initial_cost','final_cost','iterations','trial_failures','covariance','isotherms')}),flush=True)
 return out
