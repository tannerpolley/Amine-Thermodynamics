def stats(rows,obs,pred,error):
 usable=[r for r in rows if r.get(pred) is not None];n=len(usable)
 if not n:return dict(n_total=len(rows),n_available=0,n_unavailable=len(rows))
 errors=[r[error] for r in usable];positive=[r for r in usable if r[obs]>0 and r[pred]>0]
 return dict(n_total=len(rows),n_available=n,n_unavailable=len(rows)-n,bias=sum(errors)/n,RMSE=math.sqrt(sum(e*e for e in errors)/n),AARD_percent=100*sum(abs(r[pred]/r[obs]-1) for r in usable if r[obs]!=0)/sum(r[obs]!=0 for r in usable),mean_signed_relative_error_percent=100*sum(r[pred]/r[obs]-1 for r in usable if r[obs]!=0)/sum(r[obs]!=0 for r in usable),mean_ln_calc_over_obs=sum(math.log(r[pred]/r[obs]) for r in positive)/len(positive) if positive else None,min_error=min(errors),max_error=max(errors),max_abs_error=max(map(abs,errors)))
