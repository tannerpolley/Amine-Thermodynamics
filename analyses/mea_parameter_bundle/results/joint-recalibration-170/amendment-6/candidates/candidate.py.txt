"""One-off launcher (#170 A6 r3): one candidate fit plus its endpoint cold replay. Disclosed scratch, not campaign code.

Usage, from the worktree root: python candidate.py tied|held
"""
import hashlib, json, math, os, subprocess, sys
from pathlib import Path

FORM = sys.argv[1]
A = Path('analyses/mea_parameter_bundle/results/joint-recalibration-170/amendment-6')
PY = '/home/tnnrpolley21/Workspaces/Engineering/Amine-Thermodynamics/tmp/a6/venv/bin/python'
LEDGER, STEP = A.parent/'amendment-4/compute-ledger.json', A/'candidates/step-ledger.json'
ALLOCATION = dict(wall_s=1500, cpu_s=12000)  # §0b: two combined-density F1 forms, fresh releases and replays
ENV = dict(os.environ, PYTHONPATH='src:analyses/mea_parameter_bundle/scripts', OMP_NUM_THREADS='1', OPENBLAS_NUM_THREADS='1', MKL_NUM_THREADS='1')
load = lambda p: json.loads(Path(p).read_text())
dump = lambda p, x: Path(p).write_text(json.dumps(x, indent=1) + '\n')


def timed(label, argv):
    clock = A/FORM/f'{label}-command-time.json'
    code = subprocess.call(['nice', '-n', '10', '/usr/bin/time', '-f', '{"whole_command_user_s":%U,"whole_command_system_s":%S,"wall_s":%e}',
                            '-o', str(clock), *argv], env=ENV)
    t = json.loads(next(line for line in clock.read_text().splitlines() if line.startswith('{')))
    step = load(STEP)
    step['calls'].append(dict(form=FORM, call=label, exit_code=code, wall_s=t['wall_s'], cpu_s=round(t['whole_command_user_s']+t['whole_command_system_s'], 2)))
    dump(STEP, step)
    print(FORM, label, 'exit', code, step['calls'][-1], flush=True)
    return code


def check(launch, need_wall, need_cpu):
    led, step = load(LEDGER), load(STEP)
    used = dict(wall_s=sum(c['wall_s'] for c in step['calls']), cpu_s=sum(c['cpu_s'] for c in step['calls']))
    prior = dict(wall_s=math.ceil(led['compute_wall_charged_s']+used['wall_s']), cpu_s=round(led['program_cpu_charge_s']+used['cpu_s'], 2))
    budget = dict(wall_s=min(ALLOCATION['wall_s']-used['wall_s'], led['compute_wall_cap_s']-prior['wall_s']),
                  cpu_s=min(ALLOCATION['cpu_s']-used['cpu_s'], led['program_cpu_limit_s']-prior['cpu_s']))
    entry = dict(launch=launch, program_prior_wall_s=prior['wall_s'], program_prior_cpu_s=prior['cpu_s'], allocation_used=used,
                 budget_wall_s=round(budget['wall_s'], 2), budget_cpu_s=round(budget['cpu_s'], 2), need_wall_s=need_wall, need_cpu_s=need_cpu,
                 admitted=budget['wall_s'] >= need_wall and budget['cpu_s'] >= need_cpu)
    step['operator_checks'].append(entry)
    dump(STEP, step)
    print(entry, flush=True)
    if not entry['admitted']:
        sys.exit(f'{launch}: allocation does not cover the launch; return to owner')
    return entry


spec = load(A/f'inputs/revision-3/{FORM}-fit.json')
release = Path(spec['release_tests_path'])
assert all(r['released'] for r in load(release)), 'release not passed'
gate = check(f'{FORM} candidate fit (mandatory)', spec['fit_wall_s']+120, spec['fit_cpu_s'])
spec.update(release_tests_sha256=hashlib.sha256(release.read_bytes()).hexdigest(), budget_wall_s=gate['budget_wall_s'], budget_cpu_s=gate['budget_cpu_s'],
            program_prior_wall_s=gate['program_prior_wall_s'], program_prior_cpu_s=gate['program_prior_cpu_s'])
dump(A/FORM/'fit-spec.json', spec)
if timed('fit', [PY, 'analyses/mea_parameter_bundle/scripts/fit_campaign.py', str(A/FORM/'fit-spec.json'), str(A/FORM/'fit'), '--operation', 'fit', '--case', 'stage6']):
    sys.exit(f'{FORM} fit failed; return to owner before another fit')

# Kiepe reference: the 39 CO2-water rows at the release start, sqrt(weight) * residual (same wheel, fixed binary).
start = spec['information_starts'][0]['id']
info = load(A/FORM/f'release/information/{start}/native.json')
rows = [r for r in info['rows'] if r['id'].startswith('kiepe-')]
assert len(rows) == 39
dump(A/FORM/'kiepe-reference.json', dict(source=str(A/FORM/f'release/information/{start}/native.json'), rows=rows,
     weighted_residuals=[math.sqrt(r['weight'])*x for r, x in zip(info['rows'], info['residuals']) if r['id'].startswith('kiepe-')]))
check(f'{FORM} endpoint cold replay (mandatory)', 180, 500)
documents = {f'F1-{FORM}': dict(record=str(A/FORM/'fit/usable-budget-endpoint.json'), raw=str(A/FORM/'fit/native.json'),
                                density_weight=spec['fit_case']['density_weight'], kiepe_reference=str(A/FORM/'kiepe-reference.json'))}
(A/FORM/'replay').mkdir(exist_ok=True)
code = timed('replay', [PY, '-c', 'import json,sys,final_evidence as f; f.main(output_dir=sys.argv[1], documents=json.loads(sys.argv[2]), declaration=json.load(open(sys.argv[3])))',
                        str(A/FORM/'replay'), json.dumps(documents), str(A/FORM/'fit-spec.json')])
print(FORM, 'REPLAY', 'PASSED' if code == 0 else 'FAILED', flush=True)
sys.exit(code)
